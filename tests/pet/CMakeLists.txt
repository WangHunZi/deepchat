cmake_minimum_required(VERSION 3.22)
project(DeepChatPetTests LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT DEFINED UPSTREAM_ROOT OR NOT DEFINED OVERLAY_ROOT)
    message(FATAL_ERROR "Provide UPSTREAM_ROOT and OVERLAY_ROOT")
endif()
set(LVGL_ROOT "${UPSTREAM_ROOT}/managed_components/lvgl__lvgl")
if(NOT EXISTS "${LVGL_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Populate locked Xiaozhi components with the reference build first")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_ROOT}" lvgl)
set(GIF_ROOT "${UPSTREAM_ROOT}/main/display/lvgl_display/gif")
add_library(UpstreamPet STATIC
    "${UPSTREAM_ROOT}/main/device_state_machine.cc"
    "${GIF_ROOT}/gifdec.c"
    "${GIF_ROOT}/lvgl_gif.cc")
target_include_directories(UpstreamPet PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(UpstreamPet PUBLIC "${UPSTREAM_ROOT}/main" "${GIF_ROOT}")
target_link_libraries(UpstreamPet PUBLIC lvgl::lvgl)
add_executable(DCPetRoleTest DCPetRoleTest.cpp)
target_include_directories(DCPetRoleTest PRIVATE "${OVERLAY_ROOT}/main" "${OVERLAY_ROOT}/main/display")
target_link_libraries(DCPetRoleTest PRIVATE UpstreamPet)
target_compile_options(DCPetRoleTest PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
foreach(case manual cancel feedback diagnostics layout gif-once gif-loop)
    add_test(NAME pet-role-${case} COMMAND DCPetRoleTest ${case})
endforeach()
